# 金蝶云星空 WebAPI SDK 使用指南

## 快速开始

### 1. 环境准备

```bash
# 安装依赖
pip install requests
```

### 2. 配置信息

编辑 `kingdee_sdk/config.py` 填入你的金蝶服务器信息：

```python
KINGDEE_CONFIG = {
    "server_url": "http://your-server-ip/K3Cloud",  # 你的服务器地址
    "acct_id": "your_acct_id",                   # 数据中心ID
    "username": "your_username",                            # 你的用户名
    "password": "your_password",                          # 你的密码
    "lcid": 2052                                   # 语言代码(2052=中文)
}
```

**获取数据中心ID的方法：**
1. 登录金蝶Web端
2. 打开浏览器开发者工具(F12)
3. 查看网络请求，找包含 `dbid=` 的请求
4. 例如：`dbid=your_acct_id`

### 3. 基础使用示例

```python
from kingdee_sdk import KingdeeClient, AuthType

# 初始化客户端
client = KingdeeClient(
    server_url="http://your-server-ip/K3Cloud",
    acct_id="your_acct_id",
    username="your_username",
    password="your_password",
    auth_type=AuthType.PASSWORD,  # 密码认证方式
    lcid=2052
)

# 登录
client.login()
print("登录成功！")

# 查询物料
result = client.execute_bill_query(
    form_id="BD_MATERIAL",                    # 物料表单ID
    field_keys="FMaterialID,FNumber,FName",   # 要查询的字段
    limit=10                                   # 最多返回10条
)
print(f"查询到 {len(result)} 条记录")
for row in result:
    print(row)

# 登出
client.logout()
```

## 常用操作示例

### 查询物料

```python
# 查询所有物料
materials = client.execute_bill_query(
    form_id="BD_MATERIAL",
    field_keys="FMaterialID,FNumber,FName,FSpecification",
    limit=100
)

# 带条件的查询
materials = client.execute_bill_query(
    form_id="BD_MATERIAL",
    field_keys="FMaterialID,FNumber,FName",
    filter_string="FNumber like 'MAT%'",  # 编号以MAT开头
    limit=50
)
```

### 查看单据详情

```python
# 根据编号查看物料详情
detail = client.view(
    form_id="BD_MATERIAL",
    data={"Number": "MAT001"}
)
print(detail)

# 根据ID查看
detail = client.view(
    form_id="BD_MATERIAL",
    data={"Id": "123456"}
)
```

### 创建单据

```python
# 创建新物料
new_material = {
    "Creator": "your_username",
    "FNumber": "MAT999",
    "FName": "测试物料",
    "FSpecification": "规格型号",
    "FMaterialGroup": {
        "FNumber": "01"  # 物料分组编号
    }
}

result = client.save(
    form_id="BD_MATERIAL",
    data=new_material
)
print(f"创建成功，单据编号: {result.get('Number')}")
```

### 审批流程

```python
# 提交单据
check_result = client.submit(
    form_id="BD_MATERIAL",
    data={"Numbers": ["MAT999"]}
)

# 审核单据
audit_result = client.audit(
    form_id="BD_MATERIAL",
    data={"Numbers": ["MAT999"]}
)

# 反审核
unaudit_result = client.unaudit(
    form_id="BD_MATERIAL",
    data={"Numbers": ["MAT999"]}
)
```

## 高级功能

### 调试模式

```python
client = KingdeeClient(
    server_url="...",
    acct_id="...",
    username="...",
    password="...",
    auth_type=AuthType.PASSWORD,
    debug=True  # 开启调试模式，查看详细请求信息
)
```

### 获取调试信息

```python
# 获取最后一次请求的详细信息
debug_info = client.get_debug_info()
print(f"请求URL: {debug_info['url']}")
print(f"请求头: {debug_info['headers']}")
print(f"请求体: {debug_info['body']}")
```

### 自动登录

```python
# 设置 auto_login=True，调用API时自动登录
client = KingdeeClient(
    server_url="...",
    acct_id="...",
    username="...",
    password="...",
    auth_type=AuthType.PASSWORD,
    auto_login=True
)

# 不需要手动调用 login()，直接调用业务API即可
result = client.execute_bill_query(...)
```

## 错误处理

```python
from kingdee_sdk import KingdeeClient, AuthType
from kingdee_sdk.exceptions import KingdeeAPIError, AuthenticationError

try:
    client = KingdeeClient(...)
    client.login()
    
    result = client.execute_bill_query(...)
    
except AuthenticationError as e:
    print(f"登录失败: {e}")
    # 检查用户名密码是否正确
    
except KingdeeAPIError as e:
    print(f"API调用失败: {e}")
    # 检查表单ID和参数是否正确
    
except Exception as e:
    print(f"未知错误: {e}")
finally:
    if client:
        client.logout()
```

## 注意事项

1. **编码问题**：确保Python文件使用UTF-8编码保存
2. **Session管理**：SDK使用 `requests.Session` 自动管理Cookie，无需手动处理
3. **并发限制**：金蝶WebAPI有并发限制，避免同时发起大量请求
4. **密码安全**：不要在代码中硬编码密码，建议使用环境变量或配置文件
